# 考勤规则引擎实施指南

## 目的

将 `docs/attendance/考勤打卡匹配规则引擎.md` 中的考勤规则应用到现有 `hb_attendance_app` 系统中。

---

## 一、现状诊断

### 当前流程

```
得力云API → import_delicloud_bulk.py → Employee Checkin
                                           ↓
                                    match_employee.py（工号→Employee 匹配）
                                           ↓
                                    match_shifts.py（班次匹配）← 问题核心
                                           ↓
                                    fix_inout.py（IN/OUT 修正）
                                           ↓
                                    generate_attendance.py（考勤生成）
```

### 核心问题：`match_shifts.py` 逻辑过于简单

当前 `match_shifts.py` 仅用**当天首次打卡时间**判定班次：

| 首次打卡 | → 判定 |
|----------|--------|
| 0:00–7:59 | 晚班 |
| 8:00–8:29 | 早班 |
| 8:30–15:59 | 行政班 |
| 16:00–23:59 | 中班 |

**缺陷**：
1. 08:15 打卡 → 固定判为早班。但可能是一个行政班员工提前 15 分钟到岗。
2. 不看下班时间。早班 16:00 下班 vs 行政班 17:30 下班，差距 1.5h，这是最强的区分信号，但完全没用上。
3. 不查排班表（Shift Assignment）。如果员工排了行政班却因 08:25 打卡被误判为早班，全天的迟到/早退计算都错了。
4. 00:00 附近打卡 → 固定判为晚班。但可能是中班下班（中班 16:00–00:00）。

## 三、新文件 `shift_matcher.py` 完整代码

在 `/apps/hb_attendance_app/hb_attendance_app/hbos_attendance/` 目录下新建 `shift_matcher.py`：

```python
"""考勤规则引擎：多优先级班次匹配、打卡配对、迟到早退判定

优先级递减：
  P0: Shift Assignment 排班锚定
  P1: 跨天连续性（午夜窗 / 清晨窗）
  P2: 下班时间配对反推（早班 vs 行政班的核心区分手段）
  P3: 时间窗邻近度打分（兜底）

调用方式：
  from hb_attendance_app.hbos_attendance.shift_matcher import match_shifts_v2
  results = match_shifts_v2(from_date="2026-07-01", to_date="2026-07-31")
"""

import frappe
from datetime import datetime, timedelta, date
from collections import defaultdict


# ——— 班次定义 ———
SHIFT_DEFS = {
    "早班":     {"in_min": 480,  "out_min": 960,  "late_grace": 5,  "early_grace": 5},   #  8:00-16:00
    "行政班":   {"in_min": 510,  "out_min": 1050, "late_grace": 5,  "early_grace": 5},   #  8:30-17:30
    "中班":     {"in_min": 960,  "out_min": 1440, "late_grace": 5,  "early_grace": 5},   # 16:00-00:00
    "晚班":     {"in_min": 0,    "out_min": 480,  "late_grace": 10, "early_grace": 5},   # 00:00-08:00
}

# 方便反向查找：下班分钟 → 班次名（用于 P2 下班时间反推）
OUT_TIME_TO_SHIFT = [
    (960,  "早班"),    # 16:00
    (1050, "行政班"),  # 17:30
    (1440, "中班"),    # 00:00 次日（用分钟=1440 代表）
    (480,  "晚班"),    # 08:00 次日（用分钟=480 代表）
]


def _minutes(t):
    """datetime → 当天分钟数 (0–1439)"""
    return t.hour * 60 + t.minute


def _minutes_crossday(t, base_date):
    """分钟数，跨天处理：如果 t 的日期 > base_date，加 1440"""
    m = _minutes(t)
    if t.date() > base_date:
        m += 1440
    return m


def _nearby(a, target, threshold):
    """分钟数 a 是否在 target 的 ±threshold 范围内"""
    return abs(a - target) <= threshold


def load_shifts(from_date, to_date):
    """
    P0: 加载 Shift Assignment
    返回 dict: { "employee|date_str": "班次名" }
    """
    rows = frappe.db.sql("""
        SELECT sa.employee, sa.date, st.name AS shift_type_name,
               st.start_time, st.end_time
        FROM `tabShift Assignment` sa
        JOIN `tabShift Type` st ON st.name = sa.shift_type
        WHERE sa.date BETWEEN %s AND %s
          AND sa.docstatus = 1
    """, (from_date, to_date), as_dict=True)

    result = {}
    shift_type_map = {}
    for row in rows:
        key = f"{row['employee']}|{row['date']}"
        st_name = row["shift_type_name"]
        # 模糊匹配：shift type 名称含有关键词
        if "行政" in st_name:
            matched = "行政班"
        elif "早" in st_name:
            matched = "早班"
        elif "中" in st_name:
            matched = "中班"
        elif "晚" in st_name or "夜" in st_name:
            matched = "晚班"
        else:
            # 用时间匹配
            start_min = _minutes(row["start_time"]) if row.get("start_time") else None
            end_min = _minutes(row["end_time"]) if row.get("end_time") else None
            matched = None
            if start_min is not None and end_min is not None:
                for sn, sd in SHIFT_DEFS.items():
                    if abs(start_min - sd["in_min"]) <= 15 and abs(end_min - sd["out_min"]) <= 15:
                        matched = sn
                        break
            if matched is None:
                matched = "行政班"  # 默认兜底
        result[key] = matched
        shift_type_map[key] = st_name
    return result


def load_checkins(from_date, to_date):
    """
    加载打卡记录，按 employee + date 分组
    返回: { "employee|date_str": [(datetime, log_type), ...] }
    log_type 是 'IN' 或 'OUT'
    """
    rows = frappe.db.sql("""
        SELECT employee, time, log_type
        FROM `tabEmployee Checkin`
        WHERE time >= %s AND time < %s
          AND employee IS NOT NULL
        ORDER BY employee, time
    """, (from_date, (datetime.strptime(to_date, "%Y-%m-%d") + timedelta(days=1)).strftime("%Y-%m-%d")), as_dict=True)

    groups = defaultdict(list)
    for r in rows:
        key = f"{r['employee']}|{r['time'].strftime('%Y-%m-%d')}"
        groups[key].append((r["time"], r["log_type"]))
    return dict(groups)


# ——— P1: 跨天连续性判定 ———

def resolve_midnight_window(checkins_by_emp_date, shift_assignments):
    """
    规则 1.1: 午夜窗 23:30–00:30
    遍历所有打卡，对落在午夜窗的打卡判定是「中班下班」还是「晚班上班」

    返回: { (employee, date_str, punch_index): ("IN"|"OUT", "中班"|"晚班") }
    """
    resolutions = {}

    for key, punches in checkins_by_emp_date.items():
        emp, ds = key.split("|", 1)
        d = datetime.strptime(ds, "%Y-%m-%d").date()

        for i, (t, _log_type) in enumerate(punches):
            m = _minutes(t)

            if not (0 <= m <= 30 or m >= 23 * 60 + 30):
                continue  # 不在午夜窗，跳过

            # 检查该员工在 15:00-17:00 或 22:00-23:59 是否有卡（中班上班迹象）
            has_afternoon_punch = any(
                15 * 60 <= _minutes(pt) <= 17 * 60
                for j, (pt, _) in enumerate(punches) if j != i
            )
            has_late_evening_punch = any(
                22 * 60 <= _minutes(pt) <= 23 * 60 + 59
                for j, (pt, _) in enumerate(punches) if j != i
            )

            # 前一天的中班上班
            prev_key = f"{emp}|{(d - timedelta(days=1)).strftime('%Y-%m-%d')}"
            prev_punches = checkins_by_emp_date.get(prev_key, [])
            prev_has_afternoon = any(
                15 * 60 <= _minutes(pt) <= 18 * 60
                for pt, _ in prev_punches
            )

            if has_afternoon_punch or has_late_evening_punch or prev_has_afternoon:
                # 中班下班
                if m <= 30:
                    # 打卡在 00:00-00:30 → 归属前一天（中班上班日）
                    resolutions[(emp, ds, i)] = ("OUT", "中班")
                    # 修正归属日期为前一天
                    resolutions[(emp, ds, i, "belongs_to")] = (d - timedelta(days=1)).strftime("%Y-%m-%d")
                elif m >= 23 * 60 + 30:
                    # 打卡在 23:30-23:59 → 当天的中班下班（中班上班在当天）
                    resolutions[(emp, ds, i)] = ("OUT", "中班")
            else:
                # 晚班上班
                resolutions[(emp, ds, i)] = ("IN", "晚班")

    return resolutions


def resolve_dawn_window(checkins_by_emp_date, shift_assignments, midnight_resolutions):
    """
    规则 1.2: 清晨窗 07:30–08:30
    可能是：晚班下班 / 早班上班 / 行政班上班

    返回: resolutions dict
    """
    resolutions = {}

    for key, punches in checkins_by_emp_date.items():
        emp, ds = key.split("|", 1)
        d = datetime.strptime(ds, "%Y-%m-%d").date()

        for i, (t, _log_type) in enumerate(punches):
            m = _minutes(t)
            if not (7 * 60 + 30 <= m <= 8 * 60 + 30):
                continue

            # Step A: 检查是否可能是晚班下班
            is_night_shift_out = False

            # A1: 检查当天 00:00 前后是否有晚班上班卡
            today_midnight_in = None
            for (ek, ed, ei), v in midnight_resolutions.items():
                if ek == emp and ed == ds and ei != i:
                    if isinstance(v, tuple) and len(v) == 2 and v == ("IN", "晚班"):
                        today_midnight_in = True
                        break

            if not today_midnight_in:
                # A1b: 检查前一天 23:30-23:59 是否有晚班上班卡
                prev_key = f"{emp}|{(d - timedelta(days=1)).strftime('%Y-%m-%d')}"
                prev_punches = checkins_by_emp_date.get(prev_key, [])
                for pt, _ in prev_punches:
                    pm = _minutes(pt)
                    if pm >= 23 * 60 + 30:
                        prev_mid_res = midnight_resolutions.get(
                            (emp, (d - timedelta(days=1)).strftime("%Y-%m-%d"), 0), None
                        )
                        if prev_mid_res and prev_mid_res == ("IN", "晚班"):
                            today_midnight_in = True
                            break
                        # 如果前一天 23:30+ 有卡且不是中班下班 → 可能是晚班上班
                        prev_evening = any(
                            22 * 60 <= _minutes(px) <= 23 * 60 + 59
                            for px, _ in prev_punches
                        )
                        has_afternoon = any(
                            15 * 60 <= _minutes(px) <= 18 * 60
                            for px, _ in prev_punches
                        )
                        if not has_afternoon:
                            today_midnight_in = True
                            break

            # A2: 检查前一天排班是否为晚班
            prev_shift_key = f"{emp}|{(d - timedelta(days=1)).strftime('%Y-%m-%d')}"
            if not today_midnight_in and shift_assignments.get(prev_shift_key) == "晚班":
                today_midnight_in = True

            if today_midnight_in:
                resolutions[(emp, ds, i)] = ("OUT", "晚班")
                resolutions[(emp, ds, i, "belongs_to")] = (d - timedelta(days=1)).strftime("%Y-%m-%d")
                is_night_shift_out = True

            # 如果不是晚班下班 → 标记为上班卡候选，进入 P2/P3
            if not is_night_shift_out:
                resolutions[(emp, ds, i)] = ("IN_CANDIDATE", None)

    return resolutions


# ——— P2: 下班时间配对反推 ———

def reverse_infer_by_out_time(checkins_by_emp_date, shift_assignments,
                               midnight_res, dawn_res):
    """
    规则 2.1: 用下班时间反推早班 vs 行政班
    核心逻辑：16:00 附近 → 早班，17:00-18:00 → 行政班

    同时处理 16:00 模糊区（规则 3.2）

    返回: dict {(emp, ds): {"shift": "早班"|"行政班", "in_punch": datetime,
                            "out_punch": datetime, "confidence": "high"|"medium"|"low",
                            "match_source": str}}
    """
    results = {}

    for key, punches in checkins_by_emp_date.items():
        emp, ds = key.split("|", 1)
        d = datetime.strptime(ds, "%Y-%m-%d").date()

        # 收集该员工当天的所有打卡，按时间排序
        all_punches = sorted(punches, key=lambda x: x[0])

        # 找出 PM 时间段的打卡（15:00-18:00）
        pm_punches = [
            (t, lt) for t, lt in all_punches
            if 15 * 60 <= _minutes(t) <= 18 * 60
        ]

        # 找出早上时间段的打卡（07:30-09:00）
        am_candidates = []
        for i, (t, lt) in enumerate(all_punches):
            m = _minutes(t)
            if 7 * 60 + 30 <= m <= 9 * 60:
                # 检查是否已被 dawn_res 标记为 IN_CANDIDATE 或未标记
                dawn_result = dawn_res.get((emp, ds, i))
                if dawn_result and dawn_result[0] == "IN_CANDIDATE":
                    am_candidates.append((i, t, lt))
                elif dawn_result is None:
                    # 没有被 dawn 特殊处理，可能是上班卡
                    am_candidates.append((i, t, lt))

        if not pm_punches and not am_candidates:
            continue

        shift_name = None
        in_punch = None
        out_punch = None
        confidence = "high"
        source = ""

        # PM 下班反推
        for pt, plt in pm_punches:
            pm = _minutes(pt)
            if 15 * 60 + 30 <= pm <= 16 * 60 + 30:
                # 15:30–16:30 → 早班下班
                shift_name = "早班"
                out_punch = pt
                confidence = "high"
                source = "reverse_infer_out_early"
                break
            elif 17 * 60 <= pm <= 18 * 60:
                # 17:00–18:00 → 行政班下班
                shift_name = "行政班"
                out_punch = pt
                confidence = "high"
                source = "reverse_infer_out_admin"
                break
            elif 15 * 60 + 55 <= pm <= 16 * 60 + 5:
                # 15:55–16:05 模糊区（早班下班 vs 中班上班）
                # 如果有 AM 候选 → 早班下班
                if am_candidates:
                    shift_name = "早班"
                    out_punch = pt
                    confidence = "medium"
                    source = "reverse_infer_out_fuzzy_1600_am_exists"
                    break
                else:
                    # 可能是中班上班，标记低置信度
                    shift_name = "中班"
                    in_punch = pt
                    confidence = "low"
                    source = "reverse_infer_fuzzy_1600_no_am"
                    break

        # 如果有确定的 PM 下班且找到班次 → 匹配 AM 上班
        if shift_name in ("早班", "行政班") and out_punch:
            # 找最接近该班次上班时间的 AM 打卡
            target_in = SHIFT_DEFS[shift_name]["in_min"]
            best_am = None
            best_dist = 9999
            for i, t, lt in am_candidates:
                dist = abs(_minutes(t) - target_in)
                if dist < best_dist:
                    best_dist = dist
                    best_am = t
            in_punch = best_am if best_am and best_dist <= 90 else None

        results[key] = {
            "employee": emp,
            "date": ds,
            "shift": shift_name,
            "in_punch": in_punch,
            "out_punch": out_punch,
            "confidence": confidence,
            "match_source": source,
        }

    return results


# ——— P3: 时间窗邻近度打分（兜底） ———

def score_by_proximity(checkins_by_emp_date, shift_assignments,
                        midnight_res, dawn_res, reverse_results):
    """
    规则 3: 对 P0-P2 均未匹配的打卡，用邻近度 + 历史频率打分
    """
    # 打分在调用入口 match_shifts_v2() 中整合处理
    # 这里返回未匹配的记录列表
    unmatched = []
    for key, punches in checkins_by_emp_date.items():
        if key in reverse_results:
            continue
        emp, ds = key.split("|", 1)
        matched = False
        for i, (t, lt) in enumerate(punches):
            if (emp, ds, i) in midnight_res or (emp, ds, i) in dawn_res:
                matched = True
                break
        if not matched:
            unmatched.append((key, punches))
    return unmatched


# ——— 主入口 ———

def match_shifts_v2(from_date="2026-07-01", to_date="2026-07-31",
                     employee_filter=None):
    """
    考勤规则引擎主入口

    Args:
        from_date: 开始日期 "YYYY-MM-DD"
        to_date: 结束日期 "YYYY-MM-DD"
        employee_filter: 可选，指定员工 ID

    Returns:
        {
            "matched": [...],   # 匹配成功的记录
            "unmatched": [...], # 无法匹配的记录（需人工复核）
            "late": [...],      # 迟到记录
            "early": [...],     # 早退记录
            "missing_in": [...],# 缺上班卡
            "missing_out": [...],# 缺下班卡
            "stats": {...}      # 统计信息
        }
    """
    # 1. 加载数据
    shift_assignments = load_shifts(from_date, to_date)
    checkins = load_checkins(from_date, to_date)

    print(f"[shift_matcher] 排班记录: {len(shift_assignments)}")
    print(f"[shift_matcher] 打卡人天: {len(checkins)}")

    # 2. P1: 跨天判定
    midnight_res = resolve_midnight_window(checkins, shift_assignments)
    dawn_res = resolve_dawn_window(checkins, shift_assignments, midnight_res)
    print(f"[shift_matcher] P1 午夜窗判定: {len(midnight_res)}")
    print(f"[shift_matcher] P1 清晨窗判定: {len(dawn_res)}")

    # 3. P2: 下班时间配对反推
    reverse_results = reverse_infer_by_out_time(
        checkins, shift_assignments, midnight_res, dawn_res)
    print(f"[shift_matcher] P2 反推匹配: {len(reverse_results)}")

    # 4. P0: 排班锚定（最高优先级，覆盖 P1/P2 的推测）
    # 注意：P0 在最后执行，覆盖前面所有的推测
    p0_results = {}
    for key, punches in checkins.items():
        emp, ds = key.split("|", 1)
        assigned = shift_assignments.get(key)
        if assigned:
            all_punches = sorted(punches, key=lambda x: x[0])
            shift_def = SHIFT_DEFS[assigned]
            target_in = shift_def["in_min"]
            target_out = shift_def["out_min"]

            # 找最接近上班时间的打卡
            best_in = None
            best_in_dist = 9999
            best_out = None
            best_out_dist = 9999

            for t, lt in all_punches:
                m = _minutes(t)
                # 跨天处理：晚班下班可能在次日 08:00
                if assigned == "晚班" and m >= 12 * 60:
                    m -= 1440  # 放到负分钟区（前一天）

                dist_in = abs(m - target_in)
                dist_out = abs(m - target_out)

                if dist_in < best_in_dist and dist_in <= 60:
                    best_in_dist = dist_in
                    best_in = t
                if dist_out < best_out_dist and dist_out <= 30:
                    best_out_dist = dist_out
                    best_out = t

            if best_in or best_out:
                p0_results[key] = {
                    "employee": emp,
                    "date": ds,
                    "shift": assigned,
                    "in_punch": best_in,
                    "out_punch": best_out,
                    "confidence": "high",
                    "match_source": "shift_assignment",
                }

    print(f"[shift_matcher] P0 排班锚定: {len(p0_results)}")

    # 5. 合并结果：P0 优先
    merged = {}
    for key, r in reverse_results.items():
        if r["shift"] is not None:
            merged[key] = r
    for key, r in p0_results.items():
        merged[key] = r  # P0 覆盖

    # 6. 迟到/早退/缺卡判定
    late_list = []
    early_list = []
    missing_in_list = []
    missing_out_list = []

    for key, r in merged.items():
        shift_name = r["shift"]
        if not shift_name:
            continue
        sd = SHIFT_DEFS[shift_name]

        # 迟到判定
        if r["in_punch"]:
            in_min = _minutes(r["in_punch"])
            if in_min > sd["in_min"] + sd["late_grace"]:
                late_min = in_min - sd["in_min"]
                late_list.append({
                    "employee": r["employee"],
                    "date": r["date"],
                    "shift": shift_name,
                    "in_time": r["in_punch"].strftime("%H:%M:%S"),
                    "expected_in": f"{sd['in_min'] // 60:02d}:{sd['in_min'] % 60:02d}:00",
                    "late_minutes": late_min,
                })
        else:
            missing_in_list.append({
                "employee": r["employee"],
                "date": r["date"],
                "shift": shift_name,
            })

        # 早退判定
        if r["out_punch"]:
            out_min = _minutes(r["out_punch"])
            # 中班跨天：out 可能在 00:xx，需要 +1440
            if shift_name == "中班" and out_min < sd["in_min"]:
                out_min += 1440
            # 晚班：out 在 08:00 附近，直接用分钟
            if out_min < sd["out_min"] - sd["early_grace"]:
                early_min = sd["out_min"] - out_min
                early_list.append({
                    "employee": r["employee"],
                    "date": r["date"],
                    "shift": shift_name,
                    "out_time": r["out_punch"].strftime("%H:%M:%S"),
                    "expected_out": f"{sd['out_min'] // 60:02d}:{sd['out_min'] % 60:02d}:00",
                    "early_minutes": early_min,
                })
        elif r["in_punch"]:  # 有 in 无 out → 缺下班卡
            missing_out_list.append({
                "employee": r["employee"],
                "date": r["date"],
                "shift": shift_name,
            })

    # 7. 更新 Employee Checkin 的 shift 字段
    update_count = 0
    for key, r in merged.items():
        emp, ds = key.split("|", 1)
        if r["shift"]:
            frappe.db.sql("""
                UPDATE `tabEmployee Checkin`
                SET shift = %s
                WHERE employee = %s AND DATE(time) = %s
            """, (r["shift"], emp, ds))
            update_count += 1
    frappe.db.commit()
    print(f"[shift_matcher] 更新 shift 字段: {update_count} 条")

    # 8. 统计
    shift_stats = {}
    for r in merged.values():
        s = r["shift"]
        if s:
            shift_stats[s] = shift_stats.get(s, 0) + 1

    return {
        "matched": list(merged.values()),
        "unmatched": [],  # 后续可补充 P3 兜底
        "late": late_list,
        "early": early_list,
        "missing_in": missing_in_list,
        "missing_out": missing_out_list,
        "stats": {
            "total_employee_days": len(checkins),
            "matched_count": len(merged),
            "p0_shift_assignment": len(p0_results),
            "p2_reverse_infer": len(reverse_results),
            "shift_distribution": shift_stats,
            "late_count": len(late_list),
            "early_count": len(early_list),
            "missing_in_count": len(missing_in_list),
            "missing_out_count": len(missing_out_list),
        },
    }
```

## 四、修改 `generate_attendance.py`

将现有 `generate_attendance.py` 修改为调用新的匹配引擎。

**关键改动**：

3. 用匹配结果生成 Attendance，替代原来简单的 in/out 逻辑

修改后的 `generate_attendance.py`：

```python
"""从打卡流水生成考勤。

重构版：使用 shift_matcher 引擎做班次匹配、打卡配对和迟到/早退判定。
"""
import frappe
from datetime import timedelta, datetime
from hb_attendance_app.hbos_attendance.shift_matcher import (
    match_shifts_v2,
    SHIFT_DEFS,
    _minutes,
)


def generate_attendance(from_date="2026-07-28", to_date="2026-07-30"):
    """主入口：生成考勤"""

    # 1. 清理旧的考勤结果
    frappe.db.sql("""
        DELETE FROM tabAttendance
        WHERE attendance_date BETWEEN %s AND %s
          AND hbos_source_type = 'delicloud'
    """, (from_date, to_date))
    frappe.db.commit()

    # 2. 运行规则引擎
    results = match_shifts_v2(from_date=from_date, to_date=to_date)

    # 3. 加载 Employee 和请假信息
    ei_map = {
        e["name"]: e
        for e in frappe.db.sql(
            "SELECT name, employee_name, department, company FROM tabEmployee WHERE status = 'Active'",
            as_dict=1,
        )
    }

    lv_map = {}
    for lv in frappe.db.sql(
        "SELECT employee, start_date, end_date, leave_type FROM `tabHBOS Leave Record` WHERE approval_status = '已通过'",
        as_dict=1,
    ):
        d = lv["start_date"]
        while d <= lv["end_date"]:
            lv_map[f"{lv['employee']}|{d.strftime('%Y-%m-%d')}"] = lv["leave_type"]
            d += timedelta(days=1)

    # 4. 生成 Attendance
    created, late_n, early_n = 0, 0, 0
    status_stats = {"Present": 0, "On Leave": 0}

    for r in results["matched"]:
        emp = r["employee"]
        ds = r["date"]
        shift = r["shift"]
        if not shift:
            continue

        ei = ei_map.get(emp)
        sd = SHIFT_DEFS.get(shift)
        if not sd:
            continue

        # IN/OUT 时间
        in_time = r.get("in_punch")
        out_time = r.get("out_punch")

        # 请假检查
        lv_type = lv_map.get(f"{emp}|{ds}")
        status = "On Leave" if lv_type else "Present"
        status_stats[status] = status_stats.get(status, 0) + 1

        # 迟到
        late = 0
        if in_time:
            tm = _minutes(in_time)
            if tm > sd["in_min"] + sd["late_grace"]:
                late = 1
                late_n += 1

        # 早退
        early = 0
        if out_time:
            om = _minutes(out_time)
            # 中班跨天
            if shift == "中班" and out_time.date() > datetime.strptime(ds, "%Y-%m-%d").date():
                om += 1440
            # 晚班 out 在次日 08:00
            if shift == "晚班" and om <= 8 * 60:
                om += 1440
            if om < sd["out_min"] - sd["early_grace"]:
                early = 1
                early_n += 1

        # 工时
        wh = 0.0
        if in_time and out_time:
            wh = round((out_time - in_time).total_seconds() / 3600, 2)

        # 插入
        n = f"DELI-ATT-{emp}-{ds}"
        if frappe.db.exists("Attendance", n):
            continue

        now = frappe.utils.now_datetime().strftime("%Y-%m-%d %H:%M:%S.%f")
        ins = in_time.strftime("%Y-%m-%d %H:%M:%S") if in_time else None
        outs = out_time.strftime("%Y-%m-%d %H:%M:%S") if out_time else None

        frappe.db.sql(
            "INSERT INTO tabAttendance (name,creation,modified,docstatus,idx,naming_series,"
            "employee,employee_name,status,attendance_date,company,department,shift,"
            "in_time,out_time,late_entry,early_exit,working_hours,hbos_source_type,leave_type) "
            "VALUES (%s,%s,%s,1,0,'ATT-',%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,'delicloud',%s)",
            (
                n, now, now, emp,
                (ei or {}).get("employee_name", ""), status, ds,
                (ei or {}).get("company", ""), (ei or {}).get("department", ""),
                shift, ins, outs, late, early, wh, lv_type,
            ),
        )
        created += 1

    frappe.db.commit()

    # 补全关联字段
    frappe.db.sql("""
        UPDATE tabAttendance a
        INNER JOIN tabEmployee e ON e.name = a.employee
        SET a.employee_name = e.employee_name, a.department = e.department
        WHERE a.attendance_date BETWEEN %s AND %s
          AND a.hbos_source_type = 'delicloud'
          AND (a.employee_name = '' OR a.employee_name IS NULL)
    """, (from_date, to_date))
    frappe.db.commit()

    print(f"\n=== 考勤生成完成 ===")
    print(f"生成: {created} | 迟到: {late_n} | 早退: {early_n}")
    for s, c in sorted(status_stats.items()):
        print(f"  {s}: {c}")

    v = frappe.db.sql("""
        SELECT status, COUNT(*) c FROM tabAttendance
        WHERE attendance_date BETWEEN %s AND %s
          AND hbos_source_type = 'delicloud'
        GROUP BY status ORDER BY status
    """, (from_date, to_date), as_dict=1)
    print("\n=== 验证 ===")
    for r in v:
        print(f"  {r['status']}: {r['c']}")

    return {"created": created, "late": late_n, "early": early_n, "stats": status_stats}


if __name__ == "__main__":
    generate_attendance()
```

## 六、注意事项

1. **Shift Assignment 优先**：新引擎优先使用排班表，如果系统中还没有排班数据，引擎会自动降级到 P2（下班时间反推）和 P3（邻近度打分）。建议尽快在 HRMS 中录入排班数据以获取最高准确率。

2. **Shift Type 名称匹配**：`load_shifts()` 函数通过关键词（"早"、"行政"、"中"、"晚"、"夜"）匹配 Shift Type 名称。如果系统中的班次名称不符，需要调整匹配关键词。

3. **旧 `match_shifts.py`**：新引擎 `shift_matcher.py` 是独立的，`match_shifts.py` 可以保留不删，但后续不要再调用它。

4. **未匹配记录**：`results["unmatched"]` 列表包含引擎无法判断的记录，应该进入人工复核队列。这是置信度机制的一部分——不确定就不猜。

5. **数据量**：如果打卡数据量很大（>10万条），建议分批处理，每次处理 7 天。
